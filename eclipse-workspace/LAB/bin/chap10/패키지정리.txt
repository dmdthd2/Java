StringBuffer sb = new StringBuffer("abc");

StringBuilder sb2 = new StringBuilder("xyz");
sb2.insert(1, "123");
System.out.println(sb2); // x123yz

  - add(int field, int amount)
  - getTime()
- StringTokenizer
  - hasMoreTokens()

  - nextInt()

[예시]
Date date = new Date();
System.out.println(date); // 현재 날짜/시간 출력

Calendar cal = Calendar.getInstance();
System.out.println(cal.get(Calendar.YEAR)); // 현재 연도 출력

StringTokenizer st = new StringTokenizer("Java,C,Python", ",");
while(st.hasMoreTokens()) {
    System.out.println(st.nextToken());
}
// 출력: Java
//      C
//      Python

Random rand = new Random();
System.out.println(rand.nextInt(100)); // 0~99 사이 랜덤값
